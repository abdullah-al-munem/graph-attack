Step 1: git clone https://github.com/abdullah-al-munem/graph-attack.git

Step 2: git checkout stable_version

Step 3: conda create -n attack_venv python=3.10 -y 

Step 4: conda activate attack_venv

Step 5: pip install -r requirements.txt

Step 5: nvidia-smi (check cuda version)

Step 7: Install torch according to your cuda version
# CUDA 11.8
conda install pytorch==2.5.1  pytorch-cuda=11.8 -c pytorch -c nvidia
# CUDA 12.1
conda install pytorch==2.5.1 pytorch-cuda=12.1 -c pytorch -c nvidia
# CUDA 12.4
conda install pytorch==2.5.1 pytorch-cuda=12.4 -c pytorch -c nvidia
# CPU Only
conda install pytorch==2.5.1 cpuonly -c pytorch

Step 8: python install_reqs.py
